function calcularDiasDaData(texto) { const limpo = texto.replace(/\D/g, ''); if (limpo.length < 5 || limpo.length > 8) return null; let dia, mes, anoCompleto; const hoje = new Date(); const anoAtualCurto = hoje.getFullYear() % 100; if (limpo.length === 5 || limpo.length === 6) { // Se tiver 5 dígitos (ex: 51240), coloca o zero na frente -> 051240 const seis = limpo.padStart(6, '0'); dia = parseInt(seis.slice(0, 2), 10); mes = parseInt(seis.slice(2, 4), 10); const anoCurto = parseInt(seis.slice(4, 6), 10); anoCompleto = anoCurto > anoAtualCurto ? 1900 + anoCurto : 2000 + anoCurto; } else { // Se tiver 7 dígitos (ex: 5121940), coloca o zero na frente -> 05121940 const oito = limpo.padStart(8, '0'); dia = parseInt(oito.slice(0, 2), 10); mes = parseInt(oito.slice(2, 4), 10); anoCompleto = parseInt(oito.slice(4, 8), 10); } if (dia < 1 || dia > 31 || mes < 1 || mes > 12) return null; const dataAlvo = new Date(anoCompleto, mes - 1, dia); if (dataAlvo.getDate() !== dia || dataAlvo.getMonth() !== mes - 1) return null; const hojeZerado = new Date(hoje.getFullYear(), hoje.getMonth(), hoje.getDate()); const diffMs = Math.abs(hojeZerado - dataAlvo); return Math.floor(diffMs / (1000 * 60 * 60 * 24)); }
